Makefile ① 构建 ▶

定义交叉编译工具链、源文件列表与链接规则。SRCS_S 里包含 boot/start.S、lib/string.S；SRCS_C 里包含 kernel/main.c。make 把它们分别编译成 .o， 再按 kernel/linker.ld 链接成 build/minios.elf， QEMU -kernel 加载后从 _start 开始执行——这就是下面 “①构建→②启动”这条箭头的真实含义。

boot/start.S ② 启动 ▶

真正的入口标号是 _start（.globl _start）。它先设栈， 再 bl clear_bss（清零 .bss，同文件内的标号）， 最后 bl kernel_main 跳进 kernel/main.c—— 这就是“②启动→③内核入口”这条箭头。

    .section .text.boot, "ax"
    .globl _start

_start:
    /* 设置内核栈。栈从高地址向低地址增长。 */
    la.global   $sp, boot_stack_top

    /* 清零 .bss，保证未初始化的全局变量从 0 开始。 */
    bl          clear_bss

    /* 进入 C 语言内核主函数。 */
    bl          kernel_main

halt:
    idle        0
    b           halt

clear_bss:
    la.global   $t0, __bss_start
    la.global   $t1, __bss_end

1:
    beq         $t0, $t1, 2f
    st.b        $zero, $t0, 0
    addi.d      $t0, $t0, 1
    b           1b

2:
    jr          $ra

已省略：文件末尾 exception_entry 段（第 9/11 次课新增的异常/中断入口，144 字节寄存器保存），与本课 memset/memcpy/strlen 无关。

kernel/main.c ③ 内核入口 / ⑥ 返回 ▶

kernel_main() 是 C 语言入口。它 #include "string.h" 拿到六个函数的原型（③→④“声明”），随后按调用约定把参数放进寄存器调用 memset/memcpy/strlen/memmove/ strcmp/zero_and_copy——这些符号在链接时被解析到 lib/string.S 里的实现（④→⑤“实现”）；函数返回后（⑤→⑥“返回”）， kernel_main 再调用本文件内的 print_char/ print_i64_dec（内部都是包一层 printk）把结果打印到串口。

提示：memset/memcpy/strlen 在本文件里其实不止调用一处—— 前面还有第 2 次课的 data_message 演示、第 5 次课的字符串遍历演示，下面这段 才是"第 7 次课"专门的边界测试与三个新函数（memmove/strcmp/zero_and_copy）的 真实调用验收。本课的重点是下面这一段。

#include "printk.h"
#include "string.h"
... （已省略 regs_alu.h / mem_fp.h / branch_loop.h / stack_abi.h /
  syscall.h / exception.h / irq.h，与本课无关）

void kernel_main(void)
{
    ... /* 已省略第 1–6 次课的验收代码 */

    /* ---- 第 7 次课：memset/memcpy/strlen 边界测试 ---- */
    {
        char buf1[4] = { 'X', 'X', 'X', 'X' };
        char buf2[4] = { 'X', 'X', 'X', 'X' };

        /* memset n=0：契约上不得改动任何字节 */
        memset(buf1, 'Z', 0);
        printk("memset n=0 : buf[0]='");
        print_char(buf1[0]);
        printk("' (应仍为 X，未改动)\n");

        /* memset n=1：只写第 0 字节，buf[1] 不受影响 */
        memset(buf1, 'Z', 1);
        printk("memset n=1 : buf[0]='");
        print_char(buf1[0]);
        printk("' buf[1]='");
        print_char(buf1[1]);
        printk("' (应为 Z / X)\n");

        /* memcpy n=0：契约上 dst 不得改动 */
        memcpy(buf2, "AB", 0);
        printk("memcpy n=0: dst[0]='");
        print_char(buf2[0]);
        printk("' (应仍为 X，未改动)\n");

        /* memcpy n=1：只搬第 0 字节，dst[1] 不受影响 */
        memcpy(buf2, "AB", 1);
        printk("memcpy n=1: dst[0]='");
        print_char(buf2[0]);
        printk("' dst[1]='");
        print_char(buf2[1]);
        printk("' (应为 A / X)\n");

        /* strlen 边界：空串长度为 0 */
        r = (long)strlen("");
        printk("strlen \"\"  : len = ");
        print_i64_dec(r);
        printk(" (应为 0；非空串已在第 5 次课验收)\n");
    }

    /* ---- 更复杂的例子 1：zero_and_copy + strcmp ---- */
    {
        char greet[16];

        /* zero_and_copy 内部依次 bl memset/strlen/memcpy，
         * 先清零整块 greet，再把 "hi7" 搬进去。 */
        zero_and_copy(greet, "hi7", sizeof(greet));
        printk("zero_and_copy: greet=\"");
        printk(greet);
        printk("\"\n");

        /* strcmp 验证：先清零再拷贝，天然以 '\0' 结尾 */
        if (strcmp(greet, "hi7") == 0) {
            printk("strcmp verified: greet == \"hi7\"\n");
        }
        if (strcmp(greet, "hi8") != 0) {
            printk("strcmp verified: greet != \"hi8\"\n");
        }
    }

    /* ---- 更复杂的例子 2：memmove，memcpy 处理不了重叠区间 ---- */
    {
        char left[8]  = { 'A', 'B', 'C', 'D', 'E', 'F', 'G', 'H' };
        char right[8] = { 'A', 'B', 'C', 'D', 'E', 'F', 'G', 'H' };
        int i;

        /* dst < src：整体左移 2 位，覆盖区间重叠 */
        memmove(left, left + 2, 6);
        printk("memmove left  (dst<src): ");
        for (i = 0; i < 8; i++) {
            print_char(left[i]);
        }
        printk(" (期望 CDEFGHGH)\n");

        /* dst > src：整体右移 2 位，覆盖区间同样重叠，方向相反 */
        memmove(right + 2, right, 6);
        printk("memmove right (dst>src): ");
        for (i = 0; i < 8; i++) {
            print_char(right[i]);
        }
        printk(" (期望 ABABCDEF)\n");
    }

    printk("week07-libc-asm check done\n");

    ... /* 已省略第 8/9/11 次课的验收代码 */
}
include/string.h ④ 声明 ▶

只声明六个函数原型，不含实现；#include "types.h" 是为了拿到 size_t（本课不展开）。这是 kernel/main.c 能直接写 memset(...)/memmove(...) 而编译器不报错的 原因——真正的函数体在 lib/string.S 里，由链接器按同名符号接上。

#ifndef MINIOS_STRING_H
#define MINIOS_STRING_H

#include "types.h"

void *memset(void *dst, int value, size_t n);
void *memcpy(void *dst, const void *src, size_t n);
size_t strlen(const char *s);

/* memmove：区间重叠时 memcpy 处理不了，这个能。
 * strcmp：逐字节比较，返回差值。
 * zero_and_copy：组合 memset+strlen+memcpy，安全地把字符串放进固定缓冲区。
 * 实现见 lib/string.S。 */
void *memmove(void *dst, const void *src, size_t n);
int strcmp(const char *a, const char *b);
void *zero_and_copy(void *dst, const char *src, size_t dst_size);

#endif
lib/string.S ⑤ 实现 ▶

六个函数的真实汇编实现（.globl 导出符号，供链接器解析 kernel/main.c 里的调用）。memset/memcpy/ strlen/memmove/strcmp 都是叶子函数， 最后 jr $ra 直接返回；zero_and_copy 是本文件唯一的 非叶子函数——内部依次 bl memset/bl strlen/ bl memcpy，用 $s0/$s1 跨三次调用保活 dst/src。

/*
 * miniOS 最小字符串/内存工具库。
 * memset/memcpy/strlen：最基础的三个逐字节实现，第 2 次课起可用。
 * memmove：memcpy 处理不了重叠区间时的真正解法（比较地址、选方向）。
 * strcmp：逐字节比较两个字符串，返回差值。
 * zero_and_copy：组合 memset+strlen+memcpy，安全地把字符串放进固定缓冲区，
 *   是本文件里唯一的非叶子函数（内部有 bl，要保存 $ra 和跨调用存活的寄存器）。
 */

    .section .text

    .globl memset
memset:
    move        $t0, $a0          /* 保存原始 dst，作为返回值。 */
    beqz        $a2, 2f

1:
    st.b        $a1, $a0, 0
    addi.d      $a0, $a0, 1
    addi.d      $a2, $a2, -1
    bnez        $a2, 1b

2:
    move        $a0, $t0
    jr          $ra

    .globl memcpy
memcpy:
    move        $t0, $a0          /* 保存原始 dst，作为返回值。 */
    beqz        $a2, 2f

1:
    ld.bu       $t1, $a1, 0
    st.b        $t1, $a0, 0
    addi.d      $a0, $a0, 1
    addi.d      $a1, $a1, 1
    addi.d      $a2, $a2, -1
    bnez        $a2, 1b

2:
    move        $a0, $t0
    jr          $ra

/*
 * ---- memmove：memcpy 处理不了重叠区间的真正解法 ----
 * dst < src：从头往后搬，和 memcpy 一样安全（永远读在写前面）。
 * dst > src：必须从尾往前搬，否则会在读到某个字节之前就把它覆盖掉。
 * 用 sltu（无符号小于）比较两个地址——地址要按无符号理解，不能用 slt。
 */
    .globl memmove
memmove:
    move        $t0, $a0            /* 保存原始 dst，作为返回值。 */
    beqz        $a2, 4f             /* n==0：什么都不用做。 */

    sltu        $t2, $a0, $a1       /* t2 = (dst < src) ? 1 : 0 */
    bnez        $t2, 2f             /* dst < src：正向搬，和 memcpy 一样 */

    /* dst >= src：从尾往前搬，先把两个指针都移到最后一个字节 */
    add.d       $a0, $a0, $a2
    add.d       $a1, $a1, $a2
    addi.d      $a0, $a0, -1
    addi.d      $a1, $a1, -1

1:
    ld.bu       $t1, $a1, 0
    st.b        $t1, $a0, 0
    addi.d      $a0, $a0, -1
    addi.d      $a1, $a1, -1
    addi.d      $a2, $a2, -1
    bnez        $a2, 1b
    b           4f

2:
    ld.bu       $t1, $a1, 0
    st.b        $t1, $a0, 0
    addi.d      $a0, $a0, 1
    addi.d      $a1, $a1, 1
    addi.d      $a2, $a2, -1
    bnez        $a2, 2b

4:
    move        $a0, $t0
    jr          $ra

    .globl strlen
strlen:
    move        $t0, $a0          /* t0 记录字符串起始地址。 */

1:
    ld.bu       $t1, $a0, 0
    beqz        $t1, 2f
    addi.d      $a0, $a0, 1
    b           1b

2:
    sub.d       $a0, $a0, $t0
    jr          $ra

/*
 * ---- strcmp ----
 * 逐字节比较，遇到不同字节或同时读到 '\0' 就停：
 *   返回 0   ：两串相等
 *   返回 !=0 ：第一个不同字节的差值（教学简化，不做饱和处理）
 */
    .globl strcmp
strcmp:
1:
    ld.bu       $t0, $a0, 0       /* t0 = *a */
    ld.bu       $t1, $a1, 0       /* t1 = *b */
    bne         $t0, $t1, 2f      /* 字节不同：跳出去算差值 */
    beqz        $t0, 3f           /* 两边都读到 '\0'：字符串相等 */
    addi.d      $a0, $a0, 1
    addi.d      $a1, $a1, 1
    b           1b

2:
    sub.d       $a0, $t0, $t1
    jr          $ra

3:
    move        $a0, $zero
    jr          $ra

/*
 * ---- zero_and_copy（本文件唯一的非叶子函数） ----
 * 先把整块 dst 清零，再把 src 的内容搬进去。内部依次 bl memset/strlen/memcpy：
 * dst/src 要跨三次 bl 存活，借用 $s0/$s1（对照 sa_strlen_and_puts 用
 * $s0/$s1 跨两次 bl 保活 s/len 的手法，见 lib/stack_abi.S）。
 * 附带的好处：因为先 memset 清零，dst_size 只要大于 strlen(src)，
 * 结果天然是以 '\0' 结尾的合法 C 字符串——只调用 memcpy 做不到这一点。
 */
    .globl zero_and_copy
zero_and_copy:
    addi.d      $sp, $sp, -32
    st.d        $ra, $sp, 24
    st.d        $s0, $sp, 16     /* s0 = dst，跨三次 bl 保存 */
    st.d        $s1, $sp, 8      /* s1 = src，跨三次 bl 保存 */

    move        $s0, $a0         /* dst */
    move        $s1, $a1         /* src */

    move        $a1, $zero        /* memset(dst, 0, dst_size)：$a0/$a2 已是对的 */
    bl          memset

    move        $a0, $s1          /* strlen(src) */
    bl          strlen           /* a0 = len */

    move        $a2, $a0         /* memcpy(dst, src, len) */
    move        $a1, $s1
    move        $a0, $s0
    bl          memcpy           /* memcpy 返回 dst，正好也是本函数的返回值 */

    ld.d        $ra, $sp, 24
    ld.d        $s0, $sp, 16
    ld.d        $s1, $sp, 8
    addi.d      $sp, $sp, 32
    jr          $ra
